let faultState = { battery: false, signal: false, zone: false, power: false }; let troubleAcknowledge = false; let troubleBeepTimer = null; // ---------------- LOG ---------------- function log(msg, type="") { let t = new Date().toLocaleTimeString(); document.getElementById("log").innerHTML += `[${t}] ${msg}
`; } // ---------------- TROUBLE CHECK ---------------- function updateTroubleState() { let active = Object.values(faultState).some(v => v === true); if (active && !troubleAcknowledge) { setSystemState("TROUBLE"); startTroubleBeep(); log("⚠ SYSTEM TROUBLE ACTIVE", "yellow"); } if (!active) { stopTroubleBeep(); setSystemState("READY"); } } // ---------------- STATE ---------------- function setSystemState(state) { document.getElementById("state").innerText = state; } // ---------------- FAULT TRIGGERS ---------------- function triggerBatteryFault(device) { faultState.battery = true; log("BATTERY LOW: " + device, "yellow"); updateTroubleState(); } function triggerSignalFault(device) { faultState.signal = true; log("SIGNAL LOSS: " + device, "yellow"); updateTroubleState(); } function triggerZoneFault(zone) { faultState.zone = true; log("ZONE FAULT: " + zone, "yellow"); updateTroubleState(); } function triggerPowerFault() { faultState.power = true; log("POWER FAILURE (backup active)", "yellow"); updateTroubleState(); } // ---------------- ACKNOWLEDGE ---------------- function acknowledgeTrouble() { troubleAcknowledge = true; stopTroubleBeep(); setSystemState("ACKNOWLEDGED TROUBLE"); log("TROUBLE ACKNOWLEDGED", "blue"); } // ---------------- TROUBLE BEEP (REALISTIC) ---------------- let audioCtx; function startTroubleBeep() { if (troubleBeepTimer) return; audioCtx = new (window.AudioContext || window.webkitAudioContext)(); function beep() { let osc = audioCtx.createOscillator(); let gain = audioCtx.createGain(); osc.connect(gain); gain.connect(audioCtx.destination); osc.frequency.value = 800; // typische fault frequency osc.type = "square"; gain.gain.value = 0.03; osc.start(); setTimeout(() => osc.stop(), 120); } troubleBeepTimer = setInterval(() => { beep(); }, 45000); // realistisch: selten } function stopTroubleBeep() { clearInterval(troubleBeepTimer); troubleBeepTimer = null; }
